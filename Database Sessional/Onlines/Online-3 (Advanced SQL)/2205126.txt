CREATE OR REPLACE FUNCTION IS_READY_FOR_PROMOTION (EID IN NUMBER)
RETURN VARCHAR2 IS
result VARCHAR2(100);
hire number;
sal number;
mx number;
mn number;
av number;
flag number;
sub number;

BEGIN 
    flag :=0;
    
        SELECT  E.HIRE_DATE, E.SALARY, J.MIN_SALARY,J.MAX_SALARY
        FROM EMPLOYEES E
        JOIN JOBS J 
        ON (E.job_id= J.job_id)
        WHERE E.EMPLOYEE_ID =EID;
  
        hire := ( MONTHS_BETWEEN(SYSDATE, E.HIRE_DATE)/12 );
        sal := J.SALARY;
        mx := J.MAX_SALARY;
        mn := J.MIN_SALARY;
        av := (mx+mn)/2;

        IF (hire >=5 and sal>avg ) THEN
         result :='IS' || EID || 'READY FOR PROMOTION? YES';
        ELSE  
         result :='IS' || EID || 'READY FOR PROMOTION? NO';
        END IF;
    RETURN result;

END;

CREATE OR REPLACE PROCEDURE UPDATE_EMP_SALARY IS 
empid number;
oldsal number;
newsal number;
td date;
years number;
BEGIN 
    FOR V IN (SELECT e.EMPLOYEE_ID , e.salary,
              j.min_salary,e.COMMISSION_PCT, e.HIRE_DATE , max(e.HIRE_DATE)as today,
              avg(salary) as av,
              max(salary) as mx

              from EMPLOYEES e 
              join jobs j  
              on (e.JOB_ID = j.JOB_ID)) 
              join Departments d  
              on (e.department_id = d.department_id)

    LOOP  
        empid := v.employee_id; 
        oldsal := v.salary ;
        newsal := oldsal + (v.COMMISSION_PCT * oldsal) + 0.1 * (v.min_salary ) + 0.1 * (v.av);
        td := today;
        years := MONTHS_BETWEEN(td, v.hire_date)/12 ;

on ec.employee_id=e.employee_id 
join jobs j on e.job_id=j.job_id;
