SELECT
DEPARTMENT_ID,COUNT(*)
FROM EMPLOYEES
WHERE(SYSDATE-HIRE_DATE)<=600
GROUP BY DEPARTMENT_ID
HAVING COUNT(*)<=1;

GROUP BY DEPARTMENT_ID
HAVING COUNT(*)<=1;

//N03
SELECT
DEPARTMENT_ID,
MIN(HIRE_DATE) AS MIN_HIREDATE,
MAX(HIRE_DATE) AS MAX_HIREDATE,
ROUND(AVG(SALARY),2) AS AVG_SALARY
FROM EMPLOYEES
WHERE DEPARTMENT_ID IS NOT NULL
GROUP BY DEPARTMENT_ID
ORDER BY (MAX(HIRE_DATE)-MIN(HIRE_DATE)) ASC, AVG(SALARY) DESC;


//NO4
SELECT
(FIRST_NAME||' '||LAST_NAME) AS FULL_NAME,
(SALARY + NVL(COMMISSION_PCT,0) * SALARY)*12 AS CUR_ANN_SALARY,
SALARY + NVL(COMMISSION_PCT,0) * SALARY AS CUR_MONTHLY_SAL,
COMMISSION_PCT AS COMMISION_PERCENTAGE,
(((SALARY + NVL(COMMISSION_PCT,0) * SALARY)*12)*NVL(COMMISSION_PCT,-0.05) + (SALARY + NVL(COMMISSION_PCT,0.) * SALARY) ) AS INCREASE_SAL
FROM EMPLOYEES;

//N05

SELECT 
MANAGER_ID, COUNT(*) TOTAL,
ROUND(AVG(SALARY),0) AS AVG_SAL
FROM EMPLOYEES
WHERE MANAGER_ID IS NOT NULL
AND TO_CHAR(HIRE_DATE,'MON')='DEC' AND TO_CHAR(HIRE_DATE,'MON')='SEP' AND TO_CHAR(HIRE_DATE,'MON')='MAR'
GROUP BY MANAGER_ID
ORDER BY COUNT(*) ASC, AVG(SALARY) DESC;
