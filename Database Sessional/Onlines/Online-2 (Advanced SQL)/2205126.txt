SELECT E.EMPLOYEE_ID , E.FIRST_NAME , E.LAST_NAME , J.JOB_ID, E.SALARY AS MIN_SALARY , D.DEPARTMENT_NAME 
FROM EMPLOYEES E 
JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID 
JOIN JOBS J ON J.JOB_ID = E.JOB_ID 
WHERE E.SALARY > ALL( 
   SELECT AVG(MIN_SALARY) 
   FROM JOBS
   GROUP BY JOB_ID
)
AND (
SELECT COUNT(*) 
FROM EMPLOYEES E2 
JOIN DEPARTMENTS D2 ON (E2.DEPARTMENT_ID = D2.DEPARTMENT_ID)  )
) > 5 ;
